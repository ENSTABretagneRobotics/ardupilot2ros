cmake_minimum_required(VERSION 3.8)
project(ardupilot2ros)

find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(mavlink REQUIRED)

include_directories(SYSTEM ${EIGEN3_INCLUDE_DIRS})
include_directories(SYSTEM ${mavlink_INCLUDE_DIRS})

add_executable(${PROJECT_NAME}_node src/ardupilot2ros_node.cpp src/VectorNavInterface.cpp src/MAVLinkDevice.cpp src/Globals.cpp src/interval.cpp src/iboolean.cpp src/OSTimer.c src/OSNet.c src/OSMisc.c src/OSThread.c src/OSTime.c src/OSCore.c)
target_compile_definitions(${PROJECT_NAME}_node PRIVATE ENABLE_GETTIMEOFDAY_WIN32 DISABLE_OPENCV_SUPPORT)
ament_target_dependencies(${PROJECT_NAME}_node rclcpp geometry_msgs sensor_msgs tf2 tf2_geometry_msgs)
add_executable(velocity_conv_node src/velocity_conv_node.cpp)
ament_target_dependencies(velocity_conv_node rclcpp geometry_msgs nav_msgs tf2 tf2_geometry_msgs)
install(TARGETS ${PROJECT_NAME}_node velocity_conv_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch other DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_velocity_conv test/test_velocity_conv.py TIMEOUT 60)
endif()

ament_package()
